# RC-2 snapshot generation: recorded programs

The `compatibility/1.0.0-rc.1` snapshot's manifest and consumer CMake
project were generated by `generate_snapshot.py` below, which imports
the checker's own functions (`tools/check_compatibility_snapshots.py`)
so the emitted content matches the validator exactly. The archive
fixtures were produced by `make_snapshot_fixtures.cc`, sharing its
fixed schema identities with the frozen `archive_consumer.cc`.
Validation at generation time: the checker passed, and the snapshot
consumer project configured, built, and ran both tests green against a
staged `cmake --install` prefix.

## generate_snapshot.py

```python
#!/usr/bin/env python3
"""Generate the compatibility/<version> snapshot from current sources.

Imports the checker's own functions so the emitted manifest and CMake
project match the validator byte-for-byte. Authored inputs: the two
consumer sources and the archive fixtures (produced by the recorded
fixture generator). Run from the repository root AFTER the version
bump, then run tools/check_compatibility_snapshots.py to validate.
"""
import json
import sys
from pathlib import Path

REPO = Path.cwd()
sys.path.insert(0, str(REPO / "tools"))
from check_compatibility_snapshots import (  # noqa: E402
    aggregate_membership,
    current_symbols,
    current_version,
    DEFAULT_VERSION,
)
from header_manifest import load_header_manifest  # noqa: E402

version = current_version(DEFAULT_VERSION)
snapdir = REPO / "compatibility" / version
snapdir.mkdir(parents=True, exist_ok=False)
(snapdir / "consumer").mkdir()
(snapdir / "archives").mkdir()

manifest = load_header_manifest(REPO / "cmake/tess-headers.json")
headers = {
    "stable": sorted(manifest["stable"]),
    "optional-stable": sorted(manifest["optional-stable"]),
}
compat = sorted(set(headers["stable"]) | set(headers["optional-stable"]))

consumer_target = "tess_snapshot_consumer"
archive_target = "tess_snapshot_archive_consumer"
consumer_source = "consumer.cc"
archive_source = "archive_consumer.cc"

# The checker's exact expected CMakeLists template.
cmakelists = f"""cmake_minimum_required(VERSION 3.25)
project(tess_compatibility_consumer LANGUAGES CXX)
find_package(tess CONFIG REQUIRED)
add_executable({consumer_target} {consumer_source})
target_link_libraries({consumer_target} PRIVATE tess::tess)
add_executable({archive_target} {archive_source})
target_link_libraries({archive_target} PRIVATE tess::tess)
enable_testing()
add_test(NAME {consumer_target} COMMAND {consumer_target})
add_test(NAME {archive_target}
  COMMAND {archive_target} "${{TESS_SNAPSHOT_DIR}}"
)
"""
(snapdir / "consumer" / "CMakeLists.txt").write_text(cmakelists)

archives = [
    {
        "path": f"archives/{name}",
        "format": 1,
        "producer_version": version,
        "schema": schema,
    }
    for name, schema in (
        ("dense-two-field.tessarchive", "dense-two-field-v1"),
        ("sparse-two-field.tessarchive", "sparse-two-field-v1"),
    )
]

payload = {
    "version": version,
    "headers": headers,
    "aggregate_membership": aggregate_membership(REPO),
    "public_symbols": current_symbols(REPO, compat),
    "consumer": f"consumer/{consumer_source}",
    "archive_consumer": f"consumer/{archive_source}",
    "consumer_project": "consumer",
    "consumer_target": consumer_target,
    "archive_consumer_target": archive_target,
    "archives": archives,
}
(snapdir / "manifest.json").write_text(
    json.dumps(payload, indent=2, sort_keys=True) + "\n"
)
print(f"generated {snapdir} (fill in consumer sources + fixtures next)")
```

## make_snapshot_fixtures.cc

```cpp
// Recorded fixture generator for the compatibility snapshot: produces
// the archive-v1 fixtures the snapshot's manifest lists, using the
// same fixed schema identities the frozen archive_consumer declares.
#include <tess/tess.h>

#include <cstdio>
#include <fstream>
#include <vector>

namespace {
struct PassableTag {};
struct CostTag {};
using Schema = tess::FieldSchema<tess::Field<PassableTag, bool>,
                                 tess::Field<CostTag, std::uint8_t>>;
using Shape = tess::Shape<tess::Extent3{32, 32, 1}, tess::Extent3{16, 16, 1}>;
using DenseWorld = tess::AlwaysResidentWorld<Shape, Schema>;
using SparseWorld = tess::SparseResidentWorld<Shape, Schema>;
using Archive = tess::PersistenceSchema<
    0x64656E73652D3176ULL, 1,
    tess::PersistedField<PassableTag, 0x70617373ULL, 1>,
    tess::PersistedField<CostTag, 0x636f7374ULL, 1>>;

void write_file(const char* path, const std::vector<std::byte>& bytes) {
  std::ofstream out(path, std::ios::binary);
  out.write(reinterpret_cast<const char*>(bytes.data()),
            static_cast<std::streamsize>(bytes.size()));
}
}  // namespace

int main() {
  {
    DenseWorld world;
    for (auto& page : world.chunks()) {
      auto open = page.template field_span<PassableTag>();
      auto cost = page.template field_span<CostTag>();
      for (std::size_t i = 0; i < open.size(); ++i) {
        open[i] = true;
        cost[i] = 1;
      }
    }
    world.field<PassableTag>(tess::Coord3{16, 10, 0}) = false;
    world.field<CostTag>(tess::Coord3{3, 3, 0}) = 9;
    std::vector<std::byte> bytes;
    if (tess::save_world_archive<Archive>(world, bytes).bytes_written == 0) {
      return 1;
    }
    write_file("dense-two-field.tessarchive", bytes);
  }
  {
    SparseWorld world{tess::ResidencyConfig{4 * SparseWorld::page_byte_size}};
    for (const auto key : {tess::ChunkKey{0}, tess::ChunkKey{1}}) {
      world.ensure_resident(key);
      auto& page = world.chunk(key);
      for (std::uint64_t i = 0; i < SparseWorld::local_tile_count; ++i) {
        page.template field<PassableTag>(tess::LocalTileId{i}) = true;
        page.template field<CostTag>(tess::LocalTileId{i}) = 1;
      }
    }
    world.field<CostTag>(tess::Coord3{3, 3, 0}) = 9;
    std::vector<std::byte> bytes;
    if (tess::save_world_archive<Archive>(world, bytes).bytes_written == 0) {
      return 1;
    }
    write_file("sparse-two-field.tessarchive", bytes);
  }
  std::printf("fixtures written\n");
  return 0;
}
```
