172 auto advance(AsyncWorkBudget budget) -> AsyncWorkStep {
173 std::uint32_t done = 0;
174 while (done < budget.max_items && state_ == AsyncResultState::Pending) {
175 ++done;
176 ++work_;
177 step();
178 }
179 auto phase = AsyncStepState::Pending;
180 if (state_ == AsyncResultState::Ready)
181 phase = AsyncStepState::Ready;
182 else if (state_ == AsyncResultState::Stale)
183 phase = AsyncStepState::Stale;
184 else if (state_ != AsyncResultState::Pending)
185 phase = AsyncStepState::Failed;
186 return {phase, done, AsyncVersion{snapshot_ ? snapshot_->profile() : 0}};
187 }

236 void finish(NavigationOutcome outcome) noexcept {
237 outcome_ = outcome;

240 auto push(NavigationLocation at, std::uint64_t cost) -> bool {
241 if (frontier_.size() == limits_.max_frontier) {
242 finish(NavigationOutcome::CapacityExceeded);
243 return false;
244 }
245 frontier_.push_back({cost, at});
246 std::push_heap(frontier_.begin(), frontier_.end(), Greater{});
247 return true;

249 auto legal(NavigationLocation at) -> bool {
250 const auto availability = snapshot_->availability(at);
251 switch (availability) {
252 case NavigationAvailability::Legal:
253 return true;
254 case NavigationAvailability::Unavailable:
255 unknown_ = true;
256 return false;
257 case NavigationAvailability::Blocked:
258 return false;
259 case NavigationAvailability::Overflow:
260 finish(NavigationOutcome::Overflow);
261 return false;
262 case NavigationAvailability::Invalid:
263 finish(NavigationOutcome::InvalidInput);
264 return false;
265 }
266 return false;
267 }
268 void step() {
269 switch (phase_) {
270 case Phase::Init: {
271 result_ = std::make_shared<NavigationResult>();
272 if (!snapshot_ || !snapshot_->valid() || goals_.empty() ||
273 (!reverse_ &&
274 (!snapshot_->contains(start_) || !snapshot_->contains(goal_)))) {
275 finish(NavigationOutcome::InvalidInput);
276 return;
277 }
278 unknown_ = !snapshot_->complete();
279 if (limits_.max_nodes == 0 || limits_.max_frontier == 0 ||
280 limits_.max_frontier > frontier_.max_size() ||
281 limits_.max_route_steps > route_.max_size() ||
282 limits_.max_nodes > labels_.max_size()) {
283 finish(NavigationOutcome::CapacityExceeded);
284 return;
285 }
286 frontier_.reserve(limits_.max_frontier);
287 if (!reverse_) route_.reserve(limits_.max_route_steps);
288 phase_ = Phase::Seed;
289 return;
290 }
291 case Phase::Seed: {
292 if (seed_ == goals_.size()) {
293 phase_ = Phase::Pop;
294 return;
295 }
296 const auto seed = goals_[seed_++];
297 if (!snapshot_->contains(seed.location) || seed.cost == infinity_) {
298 finish(NavigationOutcome::InvalidInput);
299 return;
300 }
301 if (!legal(seed.location)) {
302 if (phase_ != Phase::Publish && !reverse_)
303 finish(unknown_ ? NavigationOutcome::Indeterminate
304 : NavigationOutcome::NoRoute);
305 return;
306 }
307 auto found = labels_.find(seed.location);
308 if (found != labels_.end() && found->second.cost <= seed.cost) return;
309 if (found == labels_.end() && labels_.size() == limits_.max_nodes) {
310 finish(NavigationOutcome::CapacityExceeded);
311 return;
312 }
313 if (!push(seed.location, seed.cost)) return;
314 labels_[seed.location] = Label{seed.cost, {}, false, false};
315 return;
316 }
317 case Phase::Pop: {
318 if (frontier_.empty()) {
319 if (unresolved_overflow_ != 0) {
320 finish(NavigationOutcome::Overflow);
321 return;
322 }
323 if (reverse_ && labels_.empty()) {
324 finish(unknown_ ? NavigationOutcome::Indeterminate
325 : NavigationOutcome::NoRoute);
326 return;
327 }
328 if (reverse_) {
329 export_ = labels_.begin();
330 phase_ = Phase::Export;
331 } else
332 finish(unknown_ ? NavigationOutcome::Indeterminate
333 : NavigationOutcome::NoRoute);
334 return;
335 }
336 std::pop_heap(frontier_.begin(), frontier_.end(), Greater{});
337 const auto current = frontier_.back();
338 frontier_.pop_back();
339 auto& label = labels_.at(current.at);
340 if (label.settled || label.cost != current.cost) return;
341 label.settled = true;
342 active_ = current.at;
343 active_cost_ = label.cost;
344 if (!reverse_ && active_ == goal_) {
345 route_cost_ = label.cost;
346 reconstruction_ = goal_;
347 phase_ = Phase::Reconstruct;
348 return;
349 }
350 cursor_ = 0;
351 degree_ = snapshot_->degree(active_, reverse_);
352 phase_ = Phase::Edges;
353 return;
354 }
355 case Phase::Edges: {
356 if (cursor_ == degree_) {
357 phase_ = Phase::Pop;
358 return;
359 }
360 const auto edge = snapshot_->edge(active_, cursor_++, reverse_);
361 if ((!reverse_ && edge.from != active_) ||
362 (reverse_ && edge.to != active_) ||
363 !snapshot_->contains(edge.from) || !snapshot_->contains(edge.to) ||
364 edge.cost == infinity_) {
365 finish(NavigationOutcome::InvalidInput);
366 return;
367 }
368 switch (edge.availability) {
369 case NavigationAvailability::Blocked:
370 return;
371 case NavigationAvailability::Unavailable:
372 unknown_ = true;
373 return;
374 case NavigationAvailability::Invalid:
375 finish(NavigationOutcome::InvalidInput);
376 return;
377 case NavigationAvailability::Overflow:
378 finish(NavigationOutcome::Overflow);
379 return;
380 case NavigationAvailability::Legal:
381 break;
382 }
383 const auto next = reverse_ ? edge.from : edge.to;
384 if (!legal(next)) return;
385 auto found = labels_.find(next);
386 if (edge.cost >= infinity_ - active_cost_) {
387 // Any finite label dominates this candidate. Remember only unseen
388 // endpoints, so a later finite alternative can discharge overflow.
389 if (found != labels_.end()) return;
390 if (labels_.size() == limits_.max_nodes) {
391 finish(NavigationOutcome::CapacityExceeded);
392 return;
393 }
394 labels_.emplace(next, Label{infinity_, {}, false, false});
395 ++unresolved_overflow_;
396 return;
397 }
398 const auto cost = active_cost_ + edge.cost;
399 if (found != labels_.end() &&
400 (found->second.settled || found->second.cost <= cost))
401 return;
402 if (found == labels_.end() && labels_.size() == limits_.max_nodes) {
403 finish(NavigationOutcome::CapacityExceeded);
404 return;
405 }
406 if (!push(next, cost)) return;
407 if (found != labels_.end() && found->second.cost == infinity_)
408 --unresolved_overflow_;
409 labels_[next] = Label{cost, edge, true, false};
410 return;
411 }
412 case Phase::Reconstruct: {
413 if (reconstruction_ == start_) {
414 reverse_cursor_ = 0;
415 phase_ = Phase::Reverse;
416 return;
417 }
418 const auto& label = labels_.at(reconstruction_);
419 if (!label.has_parent || route_.size() == limits_.max_route_steps) {
420 finish(NavigationOutcome::CapacityExceeded);
421 return;
422 }
423 route_.push_back(label.parent);
424 reconstruction_ = label.parent.from;
425 return;
426 }
427 case Phase::Reverse: {
428 if (reverse_cursor_ < route_.size() / 2) {
429 std::swap(route_[reverse_cursor_],
430 route_[route_.size() - 1 - reverse_cursor_]);
431 ++reverse_cursor_;
432 } else
433 finish(NavigationOutcome::Found);
434 return;
435 }
436 case Phase::Export: {
437 if (export_ == labels_.end()) {
438 finish(NavigationOutcome::Found);
439 return;
440 }
441 const auto& [at, label] = *export_++;
442 guidance_.emplace(
443 at, NavigationGuidance{label.cost, label.parent, label.has_parent});
444 return;
445 }
446 case Phase::Publish: {
447 result_->outcome = outcome_;
448 result_->proof = unknown_ ? NavigationProof::FeasibleOnly
449 : NavigationProof::ExactSnapshot;
450 result_->snapshot = snapshot_;
451 result_->work_items = work_;
452 if (outcome_ == NavigationOutcome::Found) {
453 result_->cost = route_cost_;
454 result_->route = std::move(route_);
455 result_->guidance = std::move(guidance_);
456 }
457 state_ = AsyncResultState::Ready;
458 return;
459 }
460 }
461 }
