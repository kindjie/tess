112 [[nodiscard]] auto availability(std::uint64_t node) const noexcept
113 -> NavigationAvailability override {
114 if (!contains(node) || scale_ == 0) return NavigationAvailability::Invalid;
115 const auto resolved = world_->resolve(coordinate(node));
116 const auto* page = world_->try_chunk(resolved.chunk_key);
117 if (page == nullptr) return NavigationAvailability::Unavailable;
118 return class_type::passable(*page, resolved.local_tile_id) &&
119 class_type::entry_cost(*page, resolved.local_tile_id) != 0
120 ? NavigationAvailability::Legal
121 : NavigationAvailability::Blocked;
122 }

124 [[nodiscard]] auto degree(std::uint64_t node, bool reverse) const noexcept
125 -> std::size_t override {
126 (void)reverse;
127 if (!contains(node)) return 0;
128 std::size_t count = 0;
129 detail::for_each_regular_candidate<shape_type, step_policy>(
130 coordinate(node), [&](auto) { ++count; });
131 return count;

134 [[nodiscard]] auto edge(std::uint64_t node, std::size_t cursor,
135 bool reverse) const -> NavigationLocalEdge override {
136 if (!contains(node))
137 return {.availability = NavigationAvailability::Invalid};
138 const auto at = coordinate(node);
139 Coord3 other{};
140 std::size_t ordinal = 0;
141 bool found = false;
142 detail::for_each_regular_candidate<shape_type, step_policy>(
143 at, [&](auto candidate) {
144 if (ordinal++ == cursor) {
145 other = candidate.to;
146 found = true;
147 }
148 });
149 if (!found) return {.availability = NavigationAvailability::Invalid};
150 const auto from = reverse ? other : at;
151 const auto to = reverse ? at : other;
152 NavigationLocalEdge result{
153 .from = static_cast<std::uint64_t>(tile_key<shape_type>(from).value),
154 .to = static_cast<std::uint64_t>(tile_key<shape_type>(to).value),
155 .availability = NavigationAvailability::Blocked};
156 ordinal = 0;
157 std::uint32_t multiplier = 1;
158 detail::for_each_regular_candidate<shape_type, step_policy>(
159 from, [&](auto candidate) {
160 if (candidate.to == to) {
161 result.id = ordinal;
162 multiplier = candidate.multiplier;
163 }
164 ++ordinal;
165 });
166 const auto source = availability(result.from);
167 if (source != NavigationAvailability::Legal) {
168 result.availability = source;
169 return result;
170 }
171 if (!filter_(*world_, from, to)) return result;
172 const auto destination = availability(result.to);
173 if (destination != NavigationAvailability::Legal) {
174 result.availability = destination;
175 return result;
176 }
177 const auto clearance = model_type::regular_availability(*world_, from, to);
178 if (clearance == TransitionAvailability::Blocked) return result;
179 if (clearance == TransitionAvailability::MissingTopology) {
180 result.availability = NavigationAvailability::Unavailable;
181 return result;
182 }
183 const auto resolved = world_->resolve(to);
184 const auto* page = world_->try_chunk(resolved.chunk_key);
185 const auto entry = static_cast<std::uint64_t>(
186 class_type::entry_cost(*page, resolved.local_tile_id));
187 // The legacy probe reserves the compact infinity before common scaling.
188 // Check entry first, even for custom classes with a wider return type.
189 if (entry >= std::numeric_limits<std::uint32_t>::max()) {
190 result.availability = NavigationAvailability::Overflow;
191 return result;
192 }
193 const auto cost = entry * multiplier; // Two compact factors fit uint64.
194 constexpr auto infinity = std::numeric_limits<std::uint64_t>::max();
195 if (cost >= std::numeric_limits<std::uint32_t>::max() ||
196 cost > (infinity - 1) / scale_) {
197 result.availability = NavigationAvailability::Overflow;
198 return result;
199 }
200 result.cost = cost * scale_;
201 result.availability = NavigationAvailability::Legal;
202 return result;
203 }
204
205 [[nodiscard]] auto complete() const noexcept -> bool override {
206 if (!world_ || scale_ == 0) return false;
207 if constexpr (std::is_same_v<typename World::residency_type,
208 AlwaysResident>) {
209 return true;
210 } else {
211 return world_->resident_count() == World::chunk_count;
212 }
213 }
