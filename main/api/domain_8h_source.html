115class NavigationGraph final : public NavigationDomain {
116 public:
117 NavigationGraph(std::size_t nodes, std::vector<NavigationLocalEdge> edges,
118 bool complete = true)
119 : edges_(std::move(edges)),
120 forward_(nodes),
121 reverse_(nodes),
122 complete_(complete) {
123 std::map<std::uint64_t, bool> ids;
124 for (std::size_t i = 0; i < edges_.size(); ++i) {
125 const auto& e = edges_[i];
126 if (e.from >= nodes || e.to >= nodes ||
127 e.cost == std::numeric_limits<std::uint64_t>::max() ||
128 !ids.emplace(e.id, true).second) {
129 valid_ = false;
130 continue;
131 }
132 forward_[static_cast<std::size_t>(e.from)].push_back(i);
133 reverse_[static_cast<std::size_t>(e.to)].push_back(i);
134 if (e.availability == NavigationAvailability::Unavailable)
135 complete_ = false;
136 }
137 }
138 [[nodiscard]] auto valid() const noexcept -> bool override { return valid_; }
139 [[nodiscard]] auto contains(std::uint64_t node) const noexcept
140 -> bool override {
141 return valid_ && node < forward_.size();
142 }
143 [[nodiscard]] auto availability(std::uint64_t node) const
144 -> NavigationAvailability override {
145 return contains(node) ? NavigationAvailability::Legal
146 : NavigationAvailability::Invalid;
147 }
148 [[nodiscard]] auto degree(std::uint64_t node, bool reverse) const
149 -> std::size_t override {
150 return contains(node) ? (reverse ? reverse_ : forward_)[node].size() : 0;
151 }
152 [[nodiscard]] auto edge(std::uint64_t node, std::size_t cursor,
153 bool reverse) const -> NavigationLocalEdge override {
154 if (!contains(node))
155 return {.availability = NavigationAvailability::Invalid};
156 const auto& indices = (reverse ? reverse_ : forward_)[node];
157 if (cursor >= indices.size())
158 return {.availability = NavigationAvailability::Invalid};
159 return edges_[indices[cursor]];
160 }
161 [[nodiscard]] auto complete() const noexcept -> bool override {
162 return complete_;
163 }
164
165 private:
166 std::vector<NavigationLocalEdge> edges_;
167 std::vector<std::vector<std::size_t>> forward_, reverse_;
168 bool complete_;
169 bool valid_ = true;
170};

212class NavigationSnapshot {
213 public:
214 explicit NavigationSnapshot(
215 std::vector<std::shared_ptr<const NavigationDomain>> domains,
216 std::vector<NavigationConnection> connections = {},
217 std::uint64_t profile = 0)
218 : connections_(std::move(connections)), profile_(profile) {
219 for (std::size_t i = 0; i < domains.size(); ++i) {
220 auto& domain = domains[i];
221 if (!domain) {
222 reject(NavigationPreparationError::NullDomain, i, 0);
223 continue;
224 }
225 const auto id = domain->identity();
226 if (!domain->valid())
227 reject(NavigationPreparationError::InvalidDomain, i, id);
228 complete_ = complete_ && domain->complete();
229 if (!domains_.emplace(id, std::move(domain)).second)
230 reject(NavigationPreparationError::DuplicateDomain, i, id);
231 }
232 std::map<std::uint64_t, bool> ids;
233 for (std::size_t i = 0; i < connections_.size(); ++i) {
234 const auto& e = connections_[i];
235 auto error = NavigationPreparationError::None;
236 if (!contains(e.from))
237 error = NavigationPreparationError::InvalidConnectionFrom;
238 else if (!contains(e.to))
239 error = NavigationPreparationError::InvalidConnectionTo;
240 else if (e.cost == std::numeric_limits<std::uint64_t>::max())
241 error = NavigationPreparationError::ReservedConnectionCost;
242 else if (!ids.emplace(e.id, true).second)
243 error = NavigationPreparationError::DuplicateConnectionId;
244 if (error != NavigationPreparationError::None) {
245 reject(error, i, e.id);
246 continue;
247 }
248 forward_[e.from].push_back(i);
249 reverse_[e.to].push_back(i);
250 if (e.availability == NavigationAvailability::Unavailable)
251 complete_ = false;
252 }
253 }
254 [[nodiscard]] auto valid() const noexcept -> bool { return valid_; }
256 [[nodiscard]] auto diagnostic() const noexcept
257 -> const NavigationPreparationDiagnostic& {
258 return diagnostic_;
259 }
260 [[nodiscard]] auto complete() const noexcept -> bool { return complete_; }
261 [[nodiscard]] auto profile() const noexcept -> std::uint64_t {
262 return profile_;
263 }
264 [[nodiscard]] auto contains(NavigationLocation at) const noexcept -> bool {
265 const auto found = domains_.find(at.domain);
266 return found != domains_.end() && found->second->contains(at.node);
267 }
268 [[nodiscard]] auto availability(NavigationLocation at) const
269 -> NavigationAvailability {
270 const auto found = domains_.find(at.domain);
271 return found == domains_.end() ? NavigationAvailability::Invalid
272 : found->second->availability(at.node);
273 }
274 [[nodiscard]] auto degree(NavigationLocation at, bool reverse) const
275 -> std::size_t {
276 const auto& domain = domains_.at(at.domain);
277 const auto& index = reverse ? reverse_ : forward_;
278 const auto found = index.find(at);
279 const auto extra = found == index.end() ? 0 : found->second.size();
280 const auto local = domain->degree(at.node, reverse);
281 return local > std::numeric_limits<std::size_t>::max() - extra
282 ? std::numeric_limits<std::size_t>::max()
283 : local + extra;
284 }
285 [[nodiscard]] auto edge(NavigationLocation at, std::size_t cursor,
286 bool reverse) const -> NavigationTransition {
287 const auto& domain = domains_.at(at.domain);
288 const auto local = domain->degree(at.node, reverse);
289 if (cursor < local) {
290 const auto e = domain->edge(at.node, cursor, reverse);
291 return {{domain->location(e.from), domain->location(e.to), e.id, e.cost,
292 e.key, e.availability},
293 false};
294 }
295 const auto& index = reverse ? reverse_ : forward_;
296 return {connections_[index.at(at)[cursor - local]], true};
297 }
298
299 private:
300 void reject(NavigationPreparationError reason, std::size_t index,
301 std::uint64_t id) noexcept {
302 valid_ = false;
303 if (diagnostic_.reason == NavigationPreparationError::None)
304 diagnostic_ = {reason, index, id};
305 }
306 NavigationPreparationDiagnostic diagnostic_{};
307 std::map<std::uint64_t, std::shared_ptr<const NavigationDomain>> domains_;
308 std::vector<NavigationConnection> connections_;
309 std::map<NavigationLocation, std::vector<std::size_t>> forward_, reverse_;
310 std::uint64_t profile_;
311 bool valid_ = true;
312 bool complete_ = true;
313};
